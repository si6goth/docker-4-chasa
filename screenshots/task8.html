Задание 8 — Безопасность: non-root пользователь и .dockerignore
# Безопасность в Dockerfile
RUN groupadd -r appuser && useradd -r -g appuser appuser
RUN chown -R appuser:appuser /app
USER appuser
# Файл .dockerignore
__pycache__/
*.pyc
.env
.git/
Dockerfile
README.md
# Проверка пользователя в контейнере
$ docker run --rm my-flask-app whoami
appuser # НЕ root!
$ docker run --rm my-flask-app id
uid=1000(appuser) gid=1000(appuser) groups=1000(appuser)
# Проверка — секретные файлы НЕ попали в образ
$ docker run --rm my-flask-app ls -la /app
drwxr-xr-x appuser appuser 4096 .
-rw-r--r-- appuser appuser 3248 app.py ✔ только код
.env — отсутствует ✘ не в образе!
.git/ — отсутствует ✘ не в образе!
