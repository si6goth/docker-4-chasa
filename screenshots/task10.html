# Файл: .github/workflows/docker-build.yml
name: Docker Build and Push
on:
  push:
    branches: [main]
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: docker/build-push-action@v5
      - uses: docker/login-action@v3
      - uses: docker/push-action@v2
