# Проверка сетей
$ docker network ls
NETWORK ID NAME DRIVER
abc123 backend-network bridge
def456 frontend-network bridge
789xyz host host
# Порт 5432 НЕ проброшен на хост — база недоступна напрямую!
curl http://localhost:5432 # Connection refused (нет проброса порта)
curl http://localhost:5000 # OK — веб доступен
# Проверка доступности из контейнера-тестера
$ docker run --network frontend --rm alpine/curl http://flask-web:5000/health
