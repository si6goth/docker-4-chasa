Задание 7 — Переменные окружения и управление конфигурацией
# Файл .env
APP_ENV=production
SECRET_KEY=supersecretkey
DATABASE_URL=postgresql://dockeruser:dockerpass@db:5432/mydb
POSTGRES_USER=dockeruser
POSTGRES_PASSWORD=dockerpass
POSTGRES_DB=mydb
# Запуск с .env
$ docker compose up -d
[+] Running 3/3
▮ Container postgres-db Started
▮ Container flask-web Started
▮ Container pgadmin Started
# Проверка переменных в контейнере
$ docker exec flask-web env | grep APP_ENV
APP_ENV=production
$ docker exec flask-web env | grep DATABASE_URL
DATABASE_URL=postgresql://dockeruser:dockerpass@db:5432/mydb
# Проверка через API
$ curl http://localhost:5000/info
{
"environment": "production",
"hostname": "abc123def456",
"secret_key_set": true,
"database_url_configured": true,
"docker_container": "yes"
}
# Второй профиль: .env.production
$ docker compose --env-file .env.production up -d
